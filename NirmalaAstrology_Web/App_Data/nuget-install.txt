-- Core ASP.NET MVC 5
Install-Package Microsoft.AspNet.Mvc -Version 5.3.0
Install-Package Microsoft.AspNet.Razor -Version 3.3.0
Install-Package Microsoft.AspNet.WebPages -Version 3.3.0

-- jQuery
Install-Package jQuery -Version 3.7.1

-- Bootstrap 5 (CSS only, JS loaded via CDN)
Install-Package Bootstrap -Version 5.3.2

-- MySQL Connector for .NET (ADO.NET)
Install-Package MySql.Data -Version 8.0.33

-- JSON (usually included with MVC)
Install-Package Newtonsoft.Json -Version 13.0.3

-- CodeDom Compiler (for .NET Framework 4.8)
Install-Package Microsoft.CodeDom.Providers.DotNetCompilerPlatform -Version 4.1.0
